cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(module07_lsm_engine LANGUAGES CXX)
  set(CMAKE_CXX_STANDARD 17)
  set(CMAKE_CXX_STANDARD_REQUIRED ON)
  enable_testing()
endif()

add_executable(module07_lsm_engine main.cpp)
if(NOT CMAKE_CXX_STANDARD)
  set_property(TARGET module07_lsm_engine PROPERTY CXX_STANDARD 17)
  set_property(TARGET module07_lsm_engine PROPERTY CXX_STANDARD_REQUIRED ON)
endif()
if(MSVC)
  target_compile_options(module07_lsm_engine PRIVATE /utf-8)
endif()
add_test(NAME module07_lsm_engine COMMAND module07_lsm_engine)
